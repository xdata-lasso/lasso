```sql
WITH RECURSIVE cnt(x) AS (
  SELECT 1
  UNION ALL
  SELECT x + 1 FROM cnt WHERE x < 333
)
INSERT INTO schools (CDSCode, StatusType, County, District, DOC, DOCType, LastUpdate, Phone, Ext)
SELECT CAST(x AS TEXT), 'Active', 'County', 'District', 'DOC', 'DOCType', '2020-01-01', 'Phone_' || x, 'Ext_' || x
FROM cnt;

WITH RECURSIVE cnt(x) AS (
  SELECT 1
  UNION ALL
  SELECT x + 1 FROM cnt WHERE x < 333
)
INSERT INTO satscores (cds, rtype, enroll12, NumTstTakr, AvgScrWrite)
SELECT CAST(x AS TEXT), 'S', 100, 50, CASE WHEN x = 333 THEN NULL ELSE x END
FROM cnt;
```
