-- DATASET 3
-- Tests exact boundary around 1000 (1000 vs 1001) using multi-row inserts
INSERT INTO carriers (cid, name) VALUES (1, 888);
INSERT INTO carriers (cid, name) VALUES (2, 999);

-- Carrier 1: 1001 flights on (2022, 6, 15) -> Should be output by Gold Query
-- Carrier 2: 1000 flights on (2022, 6, 15) -> Should NOT be output by Gold Query (catches >= 1000)
WITH RECURSIVE cnt(x) AS (
  SELECT 1
  UNION ALL
  SELECT x + 1 FROM cnt WHERE x < 1001
)
INSERT INTO flights (fid, year, month_id, day_of_month, carrier_id)
SELECT x, 2022, 6, 15, 1 FROM cnt;

WITH RECURSIVE cnt(x) AS (
  SELECT 1
  UNION ALL
  SELECT x + 1 FROM cnt WHERE x < 1000
)
INSERT INTO flights (fid, year, month_id, day_of_month, carrier_id)
SELECT x + 2000, 2022, 6, 15, 2 FROM cnt;
