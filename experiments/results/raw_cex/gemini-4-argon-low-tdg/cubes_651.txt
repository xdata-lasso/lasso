('N01', 'L1', 'T', 33.3),
('N01', 'L2', 'F', 33.3),
('N01', 'L3', 'F', 33.3),
-- N02 has 3 distinct languages -> should appear before N01 due to NULLS LAST.
('N02', 'L1', 'T', 50.0),
('N02', 'L2', 'F', 25.0),
('N02', 'L3', 'F', 25.0),
-- N03 has 2 non-null languages and 1 NULL language (allowed by SQLite PK) -> COUNT(DISTINCT Language) is 2 (excluded), whereas COUNT(*) is 3.

('N03', 'L2', 'F', 50.0),
('N03', NULL, 'F', 0.0);
