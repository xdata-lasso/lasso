-- DATASET 4
-- Tests NULL handling in foreign keys (UserId, PostId) and target aggregation column (PostHistoryTypeId)
-- Also tests GROUP BY PostId vs GROUP BY UserId
INSERT INTO users (Id, Reputation, Views) VALUES (1, 100, 100);
INSERT INTO users (Id, Reputation, Views) VALUES (2, 200, 200);
INSERT INTO users (Id, Reputation, Views) VALUES (3, 300, 300);

INSERT INTO posts (Id, ViewCount) VALUES (101, 1000);
