```sql
-- DATASET 1
-- Tests exact boundary condition (n = 200 vs n = 199), single vs composite GROUP BY, and basic ordering.
-- Station 10 has 200 total trips, split across two names (100 each). Kills mutants grouping only by start_station_id.
-- Station 20 has 199 trips (boundary miss).
-- Station 30 has 200 trips (boundary hit).
INSERT INTO station (id, name, lat, long, dock_count, city, installation_date) VALUES
(10, 'Station 10A', 37.7, -122.4, 15, 'SF', '2013-08-01'),
(20, 'Station 20', 37.7, -122.4, 15, 'SF', '2013-08-01'),
(30, 'Station 30', 37.7, -122.4, 15, 'SF', '2013-08-01');

WITH RECURSIVE cnt(x) AS (SELECT 1 UNION ALL SELECT x+1 FROM cnt WHERE x < 600)
INSERT INTO trip (id, duration, start_date, start_station_name, start_station_id, end_date, end_station_name, end_station_id, bike_id, subscription_type, zip_code)
SELECT 
    x, 
    100, 
    '2014-01-01',
    CASE 
        WHEN x <= 100 THEN 'Station 10A'
        WHEN x <= 200 THEN 'Station 10B'
        WHEN x <= 399 THEN 'Station 20'
        ELSE 'Station 30'
    END,
    CASE 
        WHEN x <= 200 THEN 10
        WHEN x <= 399 THEN 20
        ELSE 30
    END,
    '2014-01-01',
    'End Station',
    10,
    1,
    'Subscriber',
    94107
FROM cnt;
```

```sql
-- DATASET 2
-- Tests NULL handling in GROUP BY, COUNT(*) vs COUNT(col), and ORDER BY ... NULLS LAST.
-- Group 1: (start_station_id = 1, start_station_name = NULL) -> 201 rows.
-- Group 2: (start_station_id = 1, start_station_name = 'Alpha') -> 200 rows.
-- Group 3: (start_station_id = 2, start_station_name = 'Beta') -> 205 rows.
-- Group 4: (start_station_id = NULL, start_station_name = 'Gamma') -> 200 rows.
-- Kills mutants using COUNT(start_station_id) or COUNT(start_station_name), mutants dropping NULLS LAST, and mutants reversing ORDER BY.
```
