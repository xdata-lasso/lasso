-- DATASET 4
-- Tests NULL handling in T-BIL with negative and zero values.
-- In SQLite, DESC places non-NULLs before NULLs. If a mutant uses NULLS FIRST or COALESCE(T-BIL, 0), it will fail when max non-null T-BIL is negative.
INSERT INTO Patient (ID, SEX, Birthday) VALUES (1, 'M', '1950-01-01');
INSERT INTO Patient (ID, SEX, Birthday) VALUES (2, 'F', '1960-02-02');
