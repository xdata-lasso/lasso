-- DATASET 4
-- Tests tie-breaking on CustomerID ASC when SUM(Consumption) is identical, and tests SUM(DISTINCT Consumption) vs SUM(Consumption)
INSERT INTO customers (CustomerID, Segment, Currency) VALUES (50, 'LAM', 'EUR');
INSERT INTO customers (CustomerID, Segment, Currency) VALUES (60, 'LAM', 'EUR');
INSERT INTO customers (CustomerID, Segment, Currency) VALUES (70, 'LAM', 'EUR');

-- Customer 50: Two identical consumptions of 10.0 -> SUM = 20.0, SUM(DISTINCT) = 10.0
INSERT INTO yearmonth (CustomerID, Date, Consumption) VALUES (50, '201203', 10.0);
INSERT INTO yearmonth (CustomerID, Date, Consumption) VALUES (50, '201204', 10.0);

-- Customer 60: Distinct consumptions 7.0 + 8.0 = 15.0 (Winner for true SUM = 15.0)
INSERT INTO yearmonth (CustomerID, Date, Consumption) VALUES (60, '201203', 7.0);
INSERT INTO yearmonth (CustomerID, Date, Consumption) VALUES (60, '201204', 8.0);

-- Customer 70: Also SUM = 15.0, ties with Customer 60, loses on CustomerID ASC (60 < 70)
