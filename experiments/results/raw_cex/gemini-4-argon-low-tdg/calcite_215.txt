```sql
-- DATASET 1
-- Tests behavior when DEPT has rows and EMP is completely empty.
-- Because COUNT(*) without GROUP BY always returns 1 row (with value 0), EXISTS is always TRUE.
-- Refutes mutants that rewrite EXISTS(SELECT COUNT(*)...) to standard EXISTS(SELECT 1...) or INNER JOIN / IN.
INSERT INTO DEPT (DEPTNO, NAME) VALUES (10, 'ACCOUNTING');
INSERT INTO DEPT (DEPTNO, NAME) VALUES (20, 'RESEARCH');
```

```sql
-- DATASET 3
-- Tests edge case numerical values in DEPTNO (0 and negative numbers) and duplicate department names.
-- Refutes mutants that use SELECT DISTINCT NAME, or assume positive IDs (e.g., DEPTNO > 0), or confuse COUNT(*) with boolean truthiness in subqueries.
INSERT INTO DEPT (DEPTNO, NAME) VALUES (0, 'ZERO_DEPT');
INSERT INTO DEPT (DEPTNO, NAME) VALUES (-10, 'NEG_DEPT');
```
