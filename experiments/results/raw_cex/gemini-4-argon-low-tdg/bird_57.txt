-- DATASET 2
WITH RECURSIVE cnt(x) AS (
  SELECT 1
  UNION ALL
  SELECT x+1 FROM cnt WHERE x < 340
)
INSERT INTO schools (CDSCode, StatusType, County, District, DOC, DOCType, LastUpdate, Phone, Ext)
SELECT 
  printf('%05d', x),
  CASE WHEN x = 333 THEN 'Merged' ELSE 'Active' END,
  'County' || (x % 5),
  'District' || (x % 10),
  '00',
  'DocType',
  '2020-01-01',
  '800-555-' || printf('%04d', x),
  CASE WHEN x = 333 THEN '9999' ELSE printf('%03d', x % 100) END
FROM cnt;

WITH RECURSIVE cnt(x) AS (
  SELECT 1
  UNION ALL
  SELECT x+1 FROM cnt WHERE x < 336
)
INSERT INTO satscores (cds, rtype, enroll12, NumTstTakr, AvgScrRead, AvgScrMath, AvgScrWrite, NumGE1500)
SELECT 
  printf('%05d', x),
  CASE WHEN x = 333 THEN 'D' ELSE 'S' END,
  100 + x,
  50 + (x % 50),
  800 - x,
  400 + x,
  CASE 
    WHEN x = 1 THEN NULL
    WHEN x IN (332, 333, 334) THEN 500
    ELSE 1000 - x 
  END,
  20
FROM cnt;

  SELECT 1
  UNION ALL
