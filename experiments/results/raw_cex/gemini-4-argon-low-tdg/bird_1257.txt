```sql
-- DATASET 9
-- Tests lexicographical comparison quirk in SQLite:
-- In SQLite, STRFTIME('%Y', ...) - STRFTIME('%Y', ...) yields an INTEGER (since '-' is an arithmetic operator).
-- An INTEGER has NO affinity, so comparing `INTEGER < '18'` converts '18' to NUMERIC (18), resulting in numeric comparison!
-- However, if a mutant omits the subtraction or formats as TEXT, or if CRP (which has TEXT affinity) is compared to 1.0 (which converts 1.0 to '1.0'):
-- For CRP: '10.0' >= '1.0' is TRUE, '0.9' >= '1.0' is FALSE, '1' >= '1.0' is FALSE (because '1' < '1.0' lexicographically!).
-- Let's include CRP = '1' and CRP = '10.0' and age = 9 (which is < 18 numerically, but '9' > '18' lexicographically if cast to text).
INSERT INTO Patient (ID, SEX, Birthday) VALUES (1, 'M', '2016-01-01');
INSERT INTO Patient (ID, SEX, Birthday) VALUES (2, 'F', '2010-01-01');
INSERT INTO Patient (ID, SEX, Birthday) VALUES (3, 'M', '2010-01-01');
INSERT INTO Laboratory (ID, Date, CRP) VALUES (1, '2023-01-01', '2.0');
INSERT INTO Laboratory (ID, Date, CRP) VALUES (2, '2023-01-01', '1');
INSERT INTO Laboratory (ID, Date, CRP) VALUES (3, '2023-01-01', '10.0');
```
