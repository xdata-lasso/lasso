```sql
-- DATASET 3
-- Tests NULL handling in ORDER BY Name DESC (in SQLite, NULLs are considered smallest, so DESC puts NULLs last).
-- Also includes a NULL in Num_of_Component to catch mutants that add WHERE Num_of_Component IS NOT NULL.
INSERT INTO furniture (Furniture_ID, Name, Num_of_Component, Market_Rate) VALUES (1, NULL, 100, 500.0);
INSERT INTO furniture (Furniture_ID, Name, Num_of_Component, Market_Rate) VALUES (2, 'Sofa', NULL, 250.0);
```
