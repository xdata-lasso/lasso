-- DATASET 2
-- Multiple airlines with NULLs and duplicates in non-key columns, plus varying row counts across tables.
-- Tests:
-- 1. Missing DISTINCT (would return 3 rows of 3 instead of 1 row of 3).
-- 2. COUNT(col) vs COUNT(*) (NULLs in Airline, Abbreviation, Country reduce count).
-- 3. COUNT(DISTINCT col) (duplicates in Airline, Abbreviation, Country reduce count).
-- 4. PARTITION BY col in OVER() (would split counts into multiple rows/values).
-- 5. Wrong table referenced (airports has 2 rows, flights has 1 row, airlines has 3 rows).
-- 6. Unintended INNER JOIN with flights (only uid 1 has flights).
INSERT INTO airlines (uid, Airline, Abbreviation, Country) VALUES (1, 'Delta', 'DL', 'USA');
INSERT INTO airlines (uid, Airline, Abbreviation, Country) VALUES (2, 'Delta', 'DL', NULL);

-- DATASET 4
-- Multiple airlines with distinct values in all columns, testing `OVER (ORDER BY ...)` window frame mutations.
-- In SQLite, `COUNT(*) OVER (ORDER BY uid)` produces running totals (1, 2, 3, 4) which `DISTINCT` keeps as 4 rows, whereas `OVER ()` produces a single row (4).
-- Also includes diverse Country values ('USA', 'UK', 'Canada') to kill hardcoded WHERE filters.
