-- DATASET 5
-- Tests SQLite lexicographical comparison nuances on TEXT column DNA vs numeric cast, and mixed matching/non-matching rows for the same patient.
INSERT INTO Patient (ID, Description) VALUES (10, NULL);
INSERT INTO Patient (ID, Description) VALUES (20, NULL);

-- Patient 10 has one failing record and one passing record

INSERT INTO Laboratory (ID, Date, DNA) VALUES (10, '2020-01-02', '6.5');

-- Patient 20 has DNA = '10' (In SQLite, TEXT '10' < '8' is TRUE because '1' < '8' lexicographically, whereas CAST(DNA AS REAL) < 8 is FALSE)
INSERT INTO Laboratory (ID, Date, DNA) VALUES (20, '2020-01-01', '10');

-- Patient 30 has DNA = '0' (Matches both text and numeric comparison)
INSERT INTO Laboratory (ID, Date, DNA) VALUES (30, '2020-01-01', '0');
