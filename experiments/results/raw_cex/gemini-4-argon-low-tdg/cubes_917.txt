```sql
-- DATASET 6
-- Tests behavior when the primary key `aid` is NULL (permitted in SQLite for non-INTEGER PRIMARY KEYs).
-- In SQLite, NULLs are considered smaller than any other value in ORDER BY.
-- Kills mutants using `WHERE aid = (SELECT MIN(aid) FROM aircraft)` (since `aid = NULL` is unknown, and `MIN(aid)` ignores NULLs).
INSERT INTO aircraft (aid, name, distance) VALUES (1, 'First Valid', 1000);
```
