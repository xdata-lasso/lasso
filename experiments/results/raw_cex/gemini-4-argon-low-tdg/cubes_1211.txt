```sql
-- DATASET 4
-- Tests NULL handling in Product_Price. In SQLite, NULLs are considered smaller than any non-NULL value, 
-- so ORDER BY Product_Price DESC places NULLs last. Catches mutants using NULLS FIRST.
INSERT INTO "Products" ("Product_ID", "Product_Type_Code", "Product_Name", "Product_Price") VALUES (1, 'A', 'Null Price Product', NULL);
INSERT INTO "Products" ("Product_ID", "Product_Type_Code", "Product_Name", "Product_Price") VALUES (2, 'B', 'Valid Price Product', 10.00);
INSERT INTO "Products" ("Product_ID", "Product_Type_Code", "Product_Name", "Product_Price") VALUES (3, 'C', 'Zero Price Product', 0.00);
```
