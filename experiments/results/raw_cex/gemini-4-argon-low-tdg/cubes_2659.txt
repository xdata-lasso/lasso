-- DATASET 3
-- Tests aggregate function mutations (e.g., AVG mutated to SUM, COUNT, MIN, MAX) and comparison mutations (meanpoints < points mutated to <=, =, >) and OR mutated to AND.
INSERT INTO club (Club_ID, name, Region, Start_year) VALUES (1, 'Club Gamma', 'East', '2000');
-- Two players with negative points (-5.0) at 'Scrum-half'.
-- AVG(-5.0, -5.0) = -5.0. (points >= 20 OR -5.0 < -5.0) is FALSE.
