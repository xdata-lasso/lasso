-- DATASET 5
-- Tests mutants that filter 'everything = 0' in the outer query instead of the inner subquery!
-- If a mutant filters 'everything = 0' in the outer query, rank_in_cat = 1 is assigned to the 'everything = 1' row in the inner query,
-- and then filtered out in the outer query, causing the entire partition to disappear instead of returning the highest 'everything = 0' row.
INSERT INTO blue (source, great, best) VALUES (100, 'Subquery_Filter_Test_1', 'Best');
INSERT INTO blue (source, great, best) VALUES (200, 'Subquery_Filter_Test_2', 'Best');

-- Source 100: Top row has everything = 1 (throw 1000). Second row has everything = 0 (throw 500).
-- Inner filter: husband 2 gets rank 1 and is returned.
-- Outer filter mutant: husband 1 gets rank 1, then outer WHERE everything = 0 eliminates husband 1, returning nothing for source 100!
INSERT INTO require (husband, source, bill, throw, everything) VALUES (1, 100, 'Top_But_Invalid', 1000.0, 1);
