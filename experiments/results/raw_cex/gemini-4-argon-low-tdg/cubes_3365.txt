```sql
-- DATASET 5
-- Tests secondary sorting and grouping when the same CHARACTER has multiple different Durations.
-- In SQLite, GROUP BY CHARACTER, duration without secondary ORDER BY column usually orders by the GROUP BY columns ('10 Days' before '20 Days').
-- Also catches mutants that GROUP BY CHARACTER only and pick the Duration corresponding to MAX(age) or MIN(age).
INSERT INTO actor (Actor_ID, Musical_ID, Name, Character, Duration, age) VALUES (1, 1, 'Eve', 'Aladdin', '20 Days', 50);
INSERT INTO actor (Actor_ID, Musical_ID, Name, Character, Duration, age) VALUES (2, 1, 'Frank', 'Aladdin', '10 Days', 20);
INSERT INTO actor (Actor_ID, Musical_ID, Name, Character, Duration, age) VALUES (3, 1, 'Grace', 'Beast', '05 Days', 90);
```

```sql
-- DATASET 6
-- Tests reverse age correspondence for the same CHARACTER with different Durations.
-- Here, the lexicographically smaller duration ('10 Days') has the larger MAX(age) (80), while '20 Days' has MAX(age) (30).
-- Together with Dataset 5, this guaranteed-kills any mutant that groups only by CHARACTER and selects duration via MAX(age) or MIN(age).
INSERT INTO actor (Actor_ID, Musical_ID, Name, Character, Duration, age) VALUES (1, 1, 'Eve', 'Aladdin', '20 Days', 30);
```

```sql
-- DATASET 7
-- Tests NULL handling in CHARACTER, age, and Musical_ID.
-- In SQLite, NULLs are considered smaller than any non-NULL value in ORDER BY ASC (so NULL CHARACTER comes first).
-- Mutants adding WHERE CHARACTER IS NOT NULL or WHERE age IS NOT NULL or INNER JOIN will fail.
INSERT INTO actor (Actor_ID, Musical_ID, Name, Character, Duration, age) VALUES (1, NULL, 'NullCharActor', NULL, 'NullCharDuration', NULL);
```
