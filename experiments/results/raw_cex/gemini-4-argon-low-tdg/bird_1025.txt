-- Tests SUM(home_team_goal + away_team_goal) vs SUM(home_team_goal) + SUM(away_team_goal) in the presence of NULL goals.
-- In SQL, 5 + NULL = NULL, whereas SUM(col1) + SUM(col2) ignores NULLs per column.
INSERT INTO Country (id, name) VALUES (1, 'Country A');

INSERT INTO League (id, country_id, name) VALUES (1, 1, 'League NullTrap');
INSERT INTO League (id, country_id, name) VALUES (2, 1, 'League Clean');

-- League 1: Match 1 has (4, 4) -> 8. Match 2 has (10, NULL) -> NULL.
-- SUM(home + away) = 8.
-- SUM(home) + SUM(away) = (4 + 10) + 4 = 18.
INSERT INTO Match (id, country_id, league_id, season, home_team_api_id, home_team_goal, away_team_goal) VALUES (1, 1, 1, '2015/2016', 100, 4, 4);
