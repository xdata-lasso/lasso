```sql
-- DATASET 3
-- Tests NULLS LAST vs default NULLS FIRST in SQLite for SUM(T1.Score) DESC.
-- Also tests INNER JOIN vs LEFT JOIN (Post 3 has no comments).
-- Post 1: Score NULL, 2 comments -> SUM(T1.Score) = NULL, COUNT = 2.
-- Post 2: Score -5, 1 comment -> SUM(T1.Score) = -5, COUNT = 1.
-- Post 3: Score 100, 0 comments -> Excluded by INNER JOIN.
-- With NULLS LAST, -5 comes before NULL in DESC order, so Post 2 is picked (outputs 1).
-- Without NULLS LAST, in SQLite DESC puts NULLs first, picking Post 1 (outputs 2).
-- With LEFT JOIN, Post 3 would be picked (outputs 0 or 1).
INSERT INTO posts (Id, Score) VALUES (1, NULL);
INSERT INTO posts (Id, Score) VALUES (2, -5);
INSERT INTO posts (Id, Score) VALUES (3, 100);
INSERT INTO comments (Id, PostId, Score) VALUES (10, 1, 10);
INSERT INTO comments (Id, PostId, Score) VALUES (11, 1, 10);
INSERT INTO comments (Id, PostId, Score) VALUES (20, 2, 10);
```
