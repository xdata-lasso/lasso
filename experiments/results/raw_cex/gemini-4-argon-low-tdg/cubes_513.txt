-- Sam (ID 20) also has 2 friends (Both 'Sam's should appear in output; kills SELECT DISTINCT name)
INSERT INTO Friend (student_id, friend_id) VALUES (20, 30);

-- Frodo (ID 30) appears 3 times as friend_id, but 0 times as student_id (Kills mutants grouping/joining on friend_id)
