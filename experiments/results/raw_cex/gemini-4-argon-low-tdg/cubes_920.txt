```sql
-- DATASET 2
-- Multiple employees with NULLs, duplicate names, duplicate salaries, negative/zero values, and uncertified employees.
-- Tests:
-- 1. Missing DISTINCT (would return 4 rows instead of 1).
-- 2. COUNT(column) vs COUNT(*) (NULLs in name and salary reduce count).
-- 3. COUNT(DISTINCT column) (duplicate names and salaries reduce count).
-- 4. Partitioning or Ordering inside OVER(...) (e.g., OVER(ORDER BY salary) produces running counts 1, 2, 3, 4).
-- 5. Unnecessary INNER JOINs with certificate/aircraft (filters out uncertified employees or multiplies rows for multi-certified employees).
-- 6. Wrong table counted (aircraft has 2 rows, certificate has 3 rows, flight has 1 row, employee has 4 rows).
INSERT INTO aircraft (aid, name, distance) VALUES (101, 'Boeing 747', 8000);
INSERT INTO aircraft (aid, name, distance) VALUES (102, 'Airbus A320', 3000);

INSERT INTO employee (eid, name, salary) VALUES (1, 'Alice', 50000.00);
INSERT INTO employee (eid, name, salary) VALUES (2, 'Alice', NULL);
INSERT INTO employee (eid, name, salary) VALUES (3, NULL, 50000.00);
INSERT INTO employee (eid, name, salary) VALUES (4, 'Bob', 0.00);

INSERT INTO certificate (eid, aid) VALUES (1, 101);
INSERT INTO certificate (eid, aid) VALUES (1, 102);
INSERT INTO certificate (eid, aid) VALUES (2, 101);

INSERT INTO flight (flno, aid, origin, destination, distance, departure_date, arrival_date, price) VALUES (10, 101, 'SFO', 'ORD', 1800, '2023-05-01', '2023-05-01', 350.00);
```
