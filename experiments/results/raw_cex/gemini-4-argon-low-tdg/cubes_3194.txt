```sql
-- DATASET 1
-- Tests behavior when the Rooms table is completely empty (COUNT(*) should return 0).
-- We cannot insert into Reservations due to the foreign key constraint on RoomId, so the database has no rows.
-- Inserting and immediately deleting a row ensures a valid executable SQL statement block resulting in an empty state.
INSERT INTO Rooms (RoomId, roomName, beds, bedType, maxOccupancy, basePrice, decor) VALUES ('TMP', 'Temp', 1, 'Queen', 2, 100, 'modern');
DELETE FROM Rooms WHERE RoomId = 'TMP';
```
