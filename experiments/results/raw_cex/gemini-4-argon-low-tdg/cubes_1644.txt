-- DATASET 4
-- Window function framing / partitioning / ordering traps.
-- Different states/cities/bnames to catch accidental PARTITION BY or ORDER BY inside OVER(...).
-- If a mutant has COUNT(*) OVER (ORDER BY city), it will produce running counts (1, 2, 3, 4) which DISTINCT won't collapse to a single value.
-- If a mutant has COUNT(*) OVER (PARTITION BY state), it will produce partition counts (1, 2, 1) which DISTINCT collapses to (1, 2).
INSERT INTO "bank" ("branch_ID", "bname", "no_of_customers", "city", "state") VALUES (1, 'Alpha', 5, 'Austin', 'TX');
