-- Tests tie-breaking on non-NULL identical T1.time values, and lexicographical string sorting vs numerical milliseconds sorting.
INSERT INTO circuits (circuitId, circuitRef, name, url) VALUES (1, 'spa', 'Spa', 'http://url1.com');
INSERT INTO seasons (year, url) VALUES (2012, 'http://s2012.com');
INSERT INTO races (raceId, year, round, circuitId, name, date, url) VALUES (348, 2012, 1, 1, 'Race 348', '2012-06-01', 'http://r348.com');

INSERT INTO drivers (driverId, driverRef, forename, surname, url) VALUES (35, 'd35', 'Numeric', 'Fastest', 'http://d35.com');

-- Lexicographically '01:40.000' < '2:00.000', even if milliseconds say otherwise.
-- Between Driver 15 and Driver 25 (both '01:40.000'), Driver 15 has smaller driverId (15 < 25) and wins, despite Driver 25 having smaller lap (1 < 9).
INSERT INTO lapTimes (raceId, driverId, lap, position, time, milliseconds) VALUES (348, 25, 1, 1, '01:40.000', 100000);
INSERT INTO lapTimes (raceId, driverId, lap, position, time, milliseconds) VALUES (348, 15, 9, 2, '01:40.000', 100000);
-- Driver 35 has smaller milliseconds (85000 < 100000), but '1:25.000' > '01:40.000' lexicographically ('1' > '0').
INSERT INTO lapTimes (raceId, driverId, lap, position, time, milliseconds) VALUES (348, 35, 1, 1, '1:25.000', 85000);
