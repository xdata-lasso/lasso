```sql
-- DATASET 1
-- Empty database (0 rows in book).
-- Note: In SQLite, window functions over an empty table return 0 rows, whereas a standard aggregate COUNT(*) without GROUP BY returns 1 row with value 0.
INSERT INTO "book" ("Book_ID", "Title", "Issues", "Writer") VALUES (1, 'Temp', 1.0, 'Temp');
DELETE FROM "book" WHERE "Book_ID" = 1;
```
