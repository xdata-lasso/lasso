-- DATASET 4
-- Tests NULL handling in GROUP BY and COUNT(*) vs COUNT(College), as well as COUNT(DISTINCT ...) mutants and unnecessary JOIN mutants.
-- Includes rows in match_season with NULL Foreign Keys (Team, Country), duplicate values in other columns (Player, Position), and NULL College.
-- Counts via COUNT(*): NULL (2), 'Kentucky' (1), 'Kansas' (3).
-- Note: In SQLite, GROUP BY treats NULL as a group. COUNT(*) for NULL college is 2, whereas COUNT(College) is 0.
-- Expected result with COUNT(*): NULL (count 2 > 1) and 'Kansas' (count 3 > 1).
-- If a mutant uses COUNT(College), NULL has count 0, making 'Kentucky' (count 1 > 0) appear in the result!
-- Also if a mutant inner joins `team`, `country`, or `player`, rows with NULL FKs or unmatched players will be dropped, altering the counts.
INSERT INTO "country" ("Country_id", "Country_name", "Capital", "Official_native_language") VALUES (1, 'USA', 'Washington', 'English');
INSERT INTO "team" ("Team_id", "Name") VALUES (1, 'Heat');
