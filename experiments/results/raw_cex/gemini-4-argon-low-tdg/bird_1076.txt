-- DATASET 3
-- Tests difference between SUM(ball_control)/COUNT(id) vs AVG(ball_control) when ball_control is NULL, and tests DISTINCT mutations via duplicate ball_control values.
INSERT INTO "Player" ("id", "player_api_id", "player_name", "player_fifa_api_id") VALUES (1, 201, 'Abdou Diallo', 2001);
INSERT INTO "Player" ("id", "player_api_id", "player_name", "player_fifa_api_id") VALUES (2, 202, 'Aaron Appindangoye', 2002);

-- Abdou Diallo: two identical non-null values (70, 70) and one NULL.
-- Gold query: SUM is 140, COUNT(id) is 3 -> 140.0 / 3 = 46.666...
-- Mutant AVG(ball_control): 140 / 2 = 70.0
-- Mutant SUM(DISTINCT ball_control): 70 / 3 = 23.333...
