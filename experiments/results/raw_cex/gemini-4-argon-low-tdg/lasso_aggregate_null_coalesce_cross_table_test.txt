```sql
-- DATASET 1
-- Tests 1-to-many relationship (a chest referenced by multiple boxes) and unreferenced chests.
-- Kills mutants that omit the join or fail to multiply chest amount by the number of referencing boxes.
INSERT INTO chests (chest_id, amount) VALUES (1, 100);
INSERT INTO chests (chest_id, amount) VALUES (2, 200);
INSERT INTO boxes (box_id, chest_id, amount) VALUES (10, 1, 10);
INSERT INTO boxes (box_id, chest_id, amount) VALUES (11, 1, 20);
```

```sql
-- DATASET 2
-- Tests duplicate amounts across and within tables.
-- Kills mutants that use UNION instead of UNION ALL, or SUM(DISTINCT amount).
INSERT INTO chests (chest_id, amount) VALUES (1, 50);
INSERT INTO boxes (box_id, chest_id, amount) VALUES (1, 1, 50);
INSERT INTO boxes (box_id, chest_id, amount) VALUES (2, 1, 50);
```

```sql
-- DATASET 3
-- Tests boxes with NULL chest_id (unlinked boxes).
-- Kills mutants that reduce the query to a single INNER JOIN (which would drop unlinked boxes).
INSERT INTO chests (chest_id, amount) VALUES (1, 100);
INSERT INTO boxes (box_id, chest_id, amount) VALUES (1, NULL, 25);
INSERT INTO boxes (box_id, chest_id, amount) VALUES (2, 1, 35);
```

```sql
-- DATASET 4
-- Tests NULL amounts in both chests and boxes.
-- Kills mutants that combine amounts using addition (b.amount + c.amount) without COALESCE, as NULL + X = NULL.
INSERT INTO chests (chest_id, amount) VALUES (1, NULL);
INSERT INTO chests (chest_id, amount) VALUES (2, 40);
INSERT INTO boxes (box_id, chest_id, amount) VALUES (1, 1, 15);
INSERT INTO boxes (box_id, chest_id, amount) VALUES (2, 2, NULL);
```

```sql
-- DATASET 5
-- Tests ID collisions where box_id equals chest_id, but boxes.chest_id points elsewhere.
-- Kills mutants with incorrect join conditions like (b.box_id = c.chest_id) or (b.box_id = b.chest_id).
INSERT INTO chests (chest_id, amount) VALUES (1, 100);
INSERT INTO chests (chest_id, amount) VALUES (2, 500);
```
