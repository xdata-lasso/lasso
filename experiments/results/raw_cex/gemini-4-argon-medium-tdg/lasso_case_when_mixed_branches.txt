-- DATASET 1

```sql
-- DATASET 2
-- Tests case sensitivity and substring matching (e.g., LIKE '%North%', LOWER(region)) as well as duplicate amounts (DISTINCT mutations).
INSERT INTO sales (sale_id, region, amount) VALUES (1, 'North', 25.0);
INSERT INTO sales (sale_id, region, amount) VALUES (2, 'North', 25.0);
INSERT INTO sales (sale_id, region, amount) VALUES (3, 'north', 50.0);
INSERT INTO sales (sale_id, region, amount) VALUES (4, 'NorthWest', 100.0);
```

```sql
-- DATASET 3
-- Tests NULL handling in both 'region' and 'amount' columns (catches IFNULL/COALESCE issues and WHERE region IS NOT NULL filters).
INSERT INTO sales (sale_id, region, amount) VALUES (1, 'North', 40.0);
INSERT INTO sales (sale_id, region, amount) VALUES (2, 'North', NULL);
INSERT INTO sales (sale_id, region, amount) VALUES (3, NULL, 60.0);
INSERT INTO sales (sale_id, region, amount) VALUES (4, 'South', NULL);
```

```sql
-- DATASET 4
-- Tests edge case where NO rows match 'North' (SUM of CASE should be 0.0, whereas a WHERE clause filtering for 'North' would yield NULL).
INSERT INTO sales (sale_id, region, amount) VALUES (1, 'South', 30.0);
INSERT INTO sales (sale_id, region, amount) VALUES (2, 'West', 70.0);
```

```sql
-- DATASET 5
-- Tests edge case where ALL rows match 'North', and includes zero and negative amounts.
INSERT INTO sales (sale_id, region, amount) VALUES (1, 'North', 100.0);
INSERT INTO sales (sale_id, region, amount) VALUES (2, 'North', -20.0);
INSERT INTO sales (sale_id, region, amount) VALUES (3, 'North', 0.0);
```

```sql
-- DATASET 6
-- Tests division by zero when total SUM(amount) is 0 (returns NULL in SQLite).
INSERT INTO sales (sale_id, region, amount) VALUES (1, 'North', 50.0);
INSERT INTO sales (sale_id, region, amount) VALUES (2, 'South', -50.0);
```

```sql
-- DATASET 7
-- Tests when 'North' rows only have NULL amounts (CASE ELSE 0 turns NULL into NULL for the 'North' row, while 'South' becomes 0, resulting in 0.0 numerator rather than NULL).
INSERT INTO sales (sale_id, region, amount) VALUES (1, 'North', NULL);
INSERT INTO sales (sale_id, region, amount) VALUES (2, 'South', 100.0);
```
