```sql
-- DATASET 2
```

```sql
-- DATASET 3
```

```sql
-- DATASET 5
-- Single row with enr = 1:
-- For a single row where X = 1, AVG(X) = SUM(X) = MIN(X) = MAX(X) = COUNT(X) = 1.
-- By using a single row with enr = 50, we kill COUNT(enr).
-- By having multiple rows in other datasets, we kill SUM, MIN, MAX.
-- Here we use negative and fractional values to catch ABS, ROUND, CAST(AS INT) mutants.
INSERT INTO "College" ("cName", "state", "enr") VALUES ('C1', 'WA', -10.5);
INSERT INTO "College" ("cName", "state", "enr") VALUES ('C2', 'OR', -20.25);
```
