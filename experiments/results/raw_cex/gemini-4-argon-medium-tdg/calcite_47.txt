```sql
-- DATASET 3
-- Tests column confusion between EMP.EMPNO and EMP.DEPTNO, and between EMP.JOB and EMP.ENAME.
INSERT INTO DEPT (DEPTNO, NAME) VALUES (10, 'KING');
INSERT INTO DEPT (DEPTNO, NAME) VALUES (20, 'PRESIDENT');

-- EMPNO=20, DEPTNO=10, ENAME='KING', JOB='PRESIDENT' -> Matches DEPT 20 (EMPNO=20 AND JOB='PRESIDENT'). Should be excluded.
INSERT INTO EMP (EMPNO, DEPTNO, ENAME, JOB, MGR, HIREDATE, SAL, COMM, SLACKER) VALUES (20, 10, 'KING', 'PRESIDENT', NULL, '2020-03-01', 5000, 0, 0);

-- EMPNO=30, DEPTNO=20, ENAME='PRESIDENT', JOB='KING' -> Does NOT match DEPT 10 (EMPNO 30 != 10) or DEPT 20 (JOB 'KING' != 'PRESIDENT').
-- However, if a mutant uses EMP.DEPTNO instead of EMP.EMPNO, or EMP.ENAME instead of EMP.JOB, this row will be wrongly excluded.
INSERT INTO EMP (EMPNO, DEPTNO, ENAME, JOB, MGR, HIREDATE, SAL, COMM, SLACKER) VALUES (30, 20, 'PRESIDENT', 'KING', 20, '2020-03-02', 6000, 0, 0);
```

```sql
-- DATASET 4
-- Tests multiple DEPT rows with the same NAME (universal vs existential quantification if mutated to != ALL / != ANY).
INSERT INTO DEPT (DEPTNO, NAME) VALUES (10, 'ENGINEER');
INSERT INTO DEPT (DEPTNO, NAME) VALUES (20, 'ENGINEER');
INSERT INTO DEPT (DEPTNO, NAME) VALUES (30, 'HR');

-- Subquery returns (10, 20). EMPNO is 10, so 10 IN (10, 20) is TRUE -> NOT IN is FALSE (excluded).
-- If mutated to EMPNO != DEPTNO inside an EXISTS, it would see DEPT 20 (10 != 20) and wrongly include the row.
INSERT INTO EMP (EMPNO, DEPTNO, ENAME, JOB, MGR, HIREDATE, SAL, COMM, SLACKER) VALUES (10, 30, 'SCOTT', 'ENGINEER', NULL, '2020-04-01', 3500, 0, 0);

-- Subquery returns (10, 20). EMPNO is 30, so 30 IN (10, 20) is FALSE -> NOT IN is TRUE (included).
INSERT INTO EMP (EMPNO, DEPTNO, ENAME, JOB, MGR, HIREDATE, SAL, COMM, SLACKER) VALUES (30, 10, 'FORD', 'ENGINEER', 10, '2020-04-02', 4500, 0, 1);
```
