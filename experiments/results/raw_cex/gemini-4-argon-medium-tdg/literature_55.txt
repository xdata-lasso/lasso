-- DATASET 5
-- Tests the complete composite NATURAL JOIN between TAKES and SECTION on (course_id, sec_id, year).
-- If a mutant joins TAKES and SECTION only on course_id (ignoring sec_id or year), it will either drop valid rows or create a Cartesian product inflating the count.
INSERT INTO department (dept_name, building, budget) VALUES (10, 100, 50000);
INSERT INTO course (course_id, title, dept_name, credits) VALUES (101, 1, 10, 3);

INSERT INTO student (ID, name, dept_name, tot_cred) VALUES (1, 11, 10, 30);
INSERT INTO student (ID, name, dept_name, tot_cred) VALUES (2, 22, 10, 30);

-- Course 101 has two sections in different years/sec_ids, both in the valid building
INSERT INTO section (course_id, sec_id, year, building, room_number, time_slot_id) VALUES (101, 1, 2022, 100, 10, 1);
INSERT INTO section (course_id, sec_id, year, building, room_number, time_slot_id) VALUES (101, 2, 2023, 100, 10, 2);

-- Student 1 only took ONE section (101, sec 1, 2022). 
-- A broken join on just course_id=101 will match both sections in SECTION, giving Student 1 a false count of 2.
INSERT INTO takes (ID, course_id, sec_id, year, grade) VALUES (1, 101, 1, 2022, 4);

-- Student 2 took one valid section, and one section with a mismatched year (2021) not present in SECTION.
-- A broken join ignoring 'year' or omitting SECTION will give Student 2 a false count of 2.
INSERT INTO takes (ID, course_id, sec_id, year, grade) VALUES (2, 101, 1, 2022, 4);
INSERT INTO takes (ID, course_id, sec_id, year, grade) VALUES (2, 101, 1, 2021, 4);
