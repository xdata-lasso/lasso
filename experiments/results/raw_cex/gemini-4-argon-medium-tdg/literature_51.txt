-- DATASET 4
-- Tests aggregate function mutations (e.g. MIN vs MAX, AVG vs SUM, COUNT vs SUM) and negative/zero values.
INSERT INTO "department" ("dept_name", "building", "budget") VALUES (1, 10, 1000);
INSERT INTO "department" ("dept_name", "building", "budget") VALUES (2, 20, 2000);

-- Dept 1: Salaries 70, 70, -30. SUM = 110 (> 100), MAX = 70 (< 75), MIN = -30.
-- Also tests if WHERE I.SALARY < 75 is used instead of HAVING MAX(I.SALARY) < 75 (handled in Dept 2).
INSERT INTO "instructor" ("ID", "name", "salary", "dept_name") VALUES (11, 111, 70, 1);
INSERT INTO "instructor" ("ID", "name", "salary", "dept_name") VALUES (12, 112, 70, 1);

-- Dept 2: Salaries 60, 60, 80. SUM = 200 (> 100), MAX = 80 (Fails < 75), MIN = 60 (< 75), AVG = 66.67 (< 75).
-- Kills mutants replacing MAX(I.SALARY) with MIN(I.SALARY) or AVG(I.SALARY).
