-- Tests basic join conditions (id = id AND country = country), mismatched countries, mismatched IDs, and the common mutant mistake of joining on support_rep_id instead of customer.id.
INSERT INTO employees (id, first_name, last_name, country) VALUES (1, 'Alice', 'Smith', 'USA');

INSERT INTO employees (id, first_name, last_name, country) VALUES (4, 'David', 'White', 'France');

-- Matches employee 1 on both id and country, but support_rep_id points to 3
INSERT INTO customers (id, support_rep_id, first_name, last_name, country) VALUES (1, 3, 'Cust1', 'Last1', 'USA');
-- Matches employee 2 on id, but country differs ('Canada' vs 'UK')
INSERT INTO customers (id, support_rep_id, first_name, last_name, country) VALUES (2, 1, 'Cust2', 'Last2', 'Canada');
-- Matches employee 3 on country ('USA'), and support_rep_id = 3, but id = 5 (does not match employee 3's id)
INSERT INTO customers (id, support_rep_id, first_name, last_name, country) VALUES (5, 3, 'Cust5', 'Last5', 'USA');
-- Matches employee 1 on support_rep_id and country, to kill mutants that group by support_rep_id and cause duplicate rows for employee 1
INSERT INTO customers (id, support_rep_id, first_name, last_name, country) VALUES (6, 1, 'Cust6', 'Last6', 'USA');
