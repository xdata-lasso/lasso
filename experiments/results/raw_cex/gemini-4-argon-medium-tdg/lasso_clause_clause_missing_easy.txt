```sql
-- DATASET 2
-- Tests NULL handling in GROUP BY and COUNT:
-- 1. NULL category (should form its own group in GROUP BY, kills WHERE category IS NOT NULL or COUNT(category) mutants).
-- 2. NULLs in other columns (kills COUNT(name), COUNT(price), COUNT(stock_quantity) mutants).
-- 3. Empty string category (ensures '' is grouped separately from NULL).
INSERT INTO Products (product_id, name, category, price, stock_quantity) VALUES (1, NULL, 'Electronics', NULL, NULL);
INSERT INTO Products (product_id, name, category, price, stock_quantity) VALUES (2, 'Phone', 'Electronics', 500.0, 5);
INSERT INTO Products (product_id, name, category, price, stock_quantity) VALUES (3, 'Mystery Item 1', NULL, 10.0, 2);
INSERT INTO Products (product_id, name, category, price, stock_quantity) VALUES (4, NULL, NULL, NULL, NULL);
INSERT INTO Products (product_id, name, category, price, stock_quantity) VALUES (5, 'Blank Category Item', '', 5.0, 1);
```
