-- DATASET 1
-- 335 rows to test LIMIT 1 OFFSET 332, ORDER BY AvgScrWrite DESC, tie-breaking on cds, NULL handling in AvgScrWrite, and JOIN condition vs frpm.
INSERT INTO schools (CDSCode, StatusType, County, District, DOC, DOCType, LastUpdate, Phone, Ext)
WITH RECURSIVE cnt(x) AS (
    SELECT 1
    UNION ALL
    SELECT x+1 FROM cnt WHERE x < 336
)
SELECT 
    printf('CDS%04d', x), 
    'Active', 'CountyA', 'DistA', '00', 'DocTypeA', '2020-01-01',
    CASE WHEN x = 333 THEN '555-0333' ELSE printf('555-%04d', x) END,
    CASE WHEN x = 333 THEN 'EXT333' ELSE printf('E%03d', x) END
FROM cnt;

INSERT INTO satscores (cds, rtype, enroll12, NumTstTakr, AvgScrWrite, AvgScrRead, AvgScrMath)
WITH RECURSIVE cnt(x) AS (
    SELECT 1
    UNION ALL
    SELECT x+1 FROM cnt WHERE x < 335
)
SELECT 
    printf('CDS%04d', x), 
    'S', 100, 50,
    CASE 
        WHEN x = 1 THEN NULL 
        WHEN x IN (332, 333, 334) THEN 500 
        ELSE 1000 - x 
    END,
    400 + x,
    300 + x
FROM cnt;

INSERT INTO frpm (CDSCode, "Academic Year")
WITH RECURSIVE cnt(x) AS (
    SELECT 1
    UNION ALL
    SELECT x+1 FROM cnt WHERE x < 330
)
SELECT printf('CDS%04d', x), '2013-14' FROM cnt;
