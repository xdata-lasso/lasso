-- DATASET 3
-- Tests NULL handling in Name and Num_of_Component.
-- In SQLite, ORDER BY Name DESC places non-NULL values first and NULLs last (unless NULLS FIRST is explicitly specified).
-- Also includes a tie on the highest Name ('Table') where the winning row has a NULL Num_of_Component.
INSERT INTO furniture (Furniture_ID, Name, Num_of_Component, Market_Rate) VALUES (1, NULL, 99, 500.0);
