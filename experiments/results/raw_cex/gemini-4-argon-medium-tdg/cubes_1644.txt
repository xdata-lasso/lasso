```sql
-- DATASET 2
-- Multiple rows in bank with NULLs, duplicate values, and unreferenced branches.
-- Tests:
-- 1. Missing DISTINCT (would return 3 rows of 3 instead of 1 row of 3).
-- 2. COUNT(col) vs COUNT(*) (NULLs in bname, no_of_customers, city, state).
-- 3. PARTITION BY col inside OVER() (duplicate city/state values vs unique).
-- 4. Unnecessary INNER JOIN with customer or loan (only 1 branch is referenced).
-- 5. Querying wrong table (customer has 2 rows, loan has 1 row, bank has 3 rows).
INSERT INTO bank (branch_ID, bname, no_of_customers, city, state) VALUES (1, 'Main', 10, 'NYC', 'NY');
INSERT INTO bank (branch_ID, bname, no_of_customers, city, state) VALUES (2, NULL, NULL, 'NYC', 'NY');
INSERT INTO bank (branch_ID, bname, no_of_customers, city, state) VALUES (3, 'West', 0, NULL, NULL);
INSERT INTO customer (cust_ID, branch_ID, cust_name, acc_type, acc_bal, no_of_loans, credit_score, state) VALUES ('C01', 1, 'Alice', 'S', 1000, 1, 700, 'NY');
INSERT INTO customer (cust_ID, branch_ID, cust_name, acc_type, acc_bal, no_of_loans, credit_score, state) VALUES ('C02', 1, 'Bob', 'C', 2000, 0, 650, 'NY');
INSERT INTO loan (loan_ID, cust_ID, branch_ID, loan_type, amount) VALUES ('L01', 'C01', '1', 'Home', 50000);
```

-- DATASET 3
