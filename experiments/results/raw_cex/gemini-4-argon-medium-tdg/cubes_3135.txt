```sql
-- DATASET 2
-- Tests NULL handling in Name (in SQLite, NULLs are considered smaller than any non-NULL value, so DESC puts NULLs last).
-- Catches mutants that use NULLS FIRST or IS NULL filters improperly.
INSERT INTO artist (Artist_ID, Name, Country, Year_Join, Age) VALUES (1, NULL, 'USA', 2000, 30);
INSERT INTO artist (Artist_ID, Name, Country, Year_Join, Age) VALUES (2, 'Charlie', 'Canada', 2005, 35);

```
