-- DATASET 3
-- Tests join condition mutations (e.g., joining on Id instead of UserId/PostId, or T1.Id = T3.UserId)

INSERT INTO users (Id, DisplayName) VALUES (2, 'Other');

INSERT INTO posts (Id, OwnerUserId) VALUES (2, 1);

-- T2.Id = 1 matches T1.Id = 1, but T2.UserId is 2 (tests T1.Id = T2.Id mutant)
INSERT INTO postHistory (Id, PostId, UserId) VALUES (1, 1, 2);

-- T2.UserId = 1 (Harlan), T2.PostId = 2
INSERT INTO postHistory (Id, PostId, UserId) VALUES (2, 2, 1);

-- T3.Id = 2 matches T2.PostId = 2 and T2.Id = 2, but T3.PostId = 1 (tests T3.Id = T2.PostId / T2.Id mutants)
-- Also T3.UserId = 1 matches T1.Id = 1 (tests T3.UserId = T1.Id mutant)
INSERT INTO votes (Id, PostId, UserId) VALUES (2, 1, 1);

-- Valid matching vote for Post 2, but T3.UserId is 2 (not Harlan)
INSERT INTO votes (Id, PostId, UserId) VALUES (3, 2, 2);
