```sql
-- DATASET 3
-- Tests NULL handling in Market_Rate.
-- In SQLite, NULLs are considered smaller than any non-NULL value in ORDER BY ASC.
-- Thus, ORDER BY market_rate LIMIT 1 returns the NULL row, whereas MIN(market_rate) ignores NULLs.
-- This kills mutants that rewrite the query to: WHERE market_rate = (SELECT MIN(market_rate) FROM furniture).
INSERT INTO furniture (Furniture_ID, Name, Num_of_Component, Market_Rate) VALUES (1, 'Mystery_Box', 1, NULL);
INSERT INTO furniture (Furniture_ID, Name, Num_of_Component, Market_Rate) VALUES (2, 'Standard_Chair', 6, 12.5);
INSERT INTO furniture (Furniture_ID, Name, Num_of_Component, Market_Rate) VALUES (3, 'Luxury_Bed', 20, 85.0);
```
