-- DATASET 4
-- Tests SUM vs MIN/AVG/COUNT, duplicate consumption values (SUM(DISTINCT)), and case/whitespace sensitivity on 'KAM'.
INSERT INTO customers (CustomerID, Segment, Currency) VALUES (1, 'KAM', 'EUR');
INSERT INTO customers (CustomerID, Segment, Currency) VALUES (2, 'KAM', 'EUR');
INSERT INTO customers (CustomerID, Segment, Currency) VALUES (3, 'kam', 'EUR');

-- Customer 1: Two identical consumption values of 16000 in 2012.
-- SUM(Consumption) = 32000 (not < 30000).
-- SUM(DISTINCT Consumption) = 16000 (< 30000).
-- AVG(Consumption) = 16000 (< 30000).
-- Catches SUM(DISTINCT ...) and AVG(...) or MIN(...).
INSERT INTO yearmonth (CustomerID, Date, Consumption) VALUES (1, '201201', 16000.0);
INSERT INTO yearmonth (CustomerID, Date, Consumption) VALUES (1, '201202', 16000.0);

-- Customer 2: One row > 30000 and one row < 0 (or small) vs MIN.
-- Here one row is 35000 and one row is 1000. SUM is 36000 (>= 30000), MIN is 1000 (< 30000), AVG is 18000 (< 30000).
INSERT INTO yearmonth (CustomerID, Date, Consumption) VALUES (2, '201203', 35000.0);
INSERT INTO yearmonth (CustomerID, Date, Consumption) VALUES (2, '201204', 1000.0);

-- Customer 3: Lowercase 'kam'. Catches case-insensitive comparisons if mutated.
INSERT INTO yearmonth (CustomerID, Date, Consumption) VALUES (3, '201201', 5000.0);
