```sql
-- DATASET 2
-- Multiple rows in performance with NULLs, duplicates in attributes, and different row counts across tables.
-- Tests:
-- 1. DISTINCT (without DISTINCT, window function returns N rows of N instead of 1 row of N).
-- 2. COUNT(*) vs COUNT(column) where columns have NULLs.
-- 3. COUNT(*) vs COUNT(DISTINCT column) where columns have duplicates.
-- 4. Unnecessary INNER JOINs with member_attendance (only 1 of 3 performances is referenced, and 1 is referenced twice).
-- 5. Querying the wrong table (member has 2 rows, performance has 3, member_attendance has 2).
INSERT INTO "member" ("Member_ID", "Name", "Nationality", "Role") VALUES ('1', 'Alice', 'USA', 'Violin');
INSERT INTO "member" ("Member_ID", "Name", "Nationality", "Role") VALUES ('2', 'Bob', 'UK', 'Cello');

INSERT INTO "performance" ("Performance_ID", "Date", "Host", "Location", "Attendance") VALUES (1, NULL, NULL, NULL, NULL);
INSERT INTO "performance" ("Performance_ID", "Date", "Host", "Location", "Attendance") VALUES (2, '2023-01-01', 'HostA', 'LocA', 100);
INSERT INTO "performance" ("Performance_ID", "Date", "Host", "Location", "Attendance") VALUES (3, '2023-01-01', 'HostA', 'LocA', 100);

INSERT INTO "member_attendance" ("Member_ID", "Performance_ID", "Num_of_Pieces") VALUES ('1', 2, 5);
INSERT INTO "member_attendance" ("Member_ID", "Performance_ID", "Num_of_Pieces") VALUES ('2', 2, 3);
```
