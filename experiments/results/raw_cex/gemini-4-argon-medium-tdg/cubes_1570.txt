-- DATASET 2
-- Tests multiple rows in 'candidate' with NULLs, duplicate values across columns, and unreferenced 'people'.
-- Catches mutants that:
-- 1. Omit DISTINCT (would return 3 rows of 3 instead of 1 row of 3).
-- 2. Use COUNT(column) or COUNT(DISTINCT column) instead of COUNT(*).
-- 3. Perform an INNER JOIN or RIGHT JOIN with 'people' (People_ID is NULL in row 2, and People_ID 3 has no candidate).
-- 4. Add a PARTITION BY or ORDER BY inside the OVER() clause.
INSERT INTO people (People_ID, Sex, Name, Date_of_Birth, Height, Weight) VALUES (1, 'M', 'Alice', '1975-03-20', 170.0, 68.0);
