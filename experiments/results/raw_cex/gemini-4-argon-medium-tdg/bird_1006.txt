-- DATASET 4
-- Tests the two-colon branch (HH:MM:SS.mmm format where INSTR(SUBSTRING(time, INSTR(time, ':') + 1), ':') > 0)
-- vs single-colon branch (MM:SS.mmm).
-- In the Gold Query's formula, a string without a second colon evaluates the first CASE to 3600*M + 60*M + S + ms/1000,
-- whereas a string with a second colon ('1:00:01.000') evaluates the first CASE to 0 and second to 60*1 = 60!
-- This dataset specifically catches any mutant that replaces the gold query's exact quirky arithmetic with standard HH:MM:SS conversion or milliseconds.
INSERT INTO circuits (circuitId, circuitRef, name, url) VALUES (1, 'baku', 'Baku', 'http://baku');
