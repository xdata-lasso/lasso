-- DATASET 3
-- Tests standard F1 1-digit minute format ('1:35.200') vs 2-digit minute format, and SUBSTRING(time, 7, 2) vs SUBSTRING(time, 7, 3).
INSERT INTO circuits (circuitId, circuitRef, name, url) VALUES (1, 'spa', 'Spa', 'http://url1');
INSERT INTO seasons (year, url) VALUES (2022, 'http://s2022');
INSERT INTO races (raceId, year, round, circuitId, name, date, url) VALUES (1, 2022, 1, 1, 'Belgian GP', '2022-08-28', 'http://r1');

INSERT INTO drivers (driverId, driverRef, forename, surname, nationality, url) VALUES (1, 'prost', 'Alain', 'Prost', 'French', 'http://d1');

-- Standard F1 format '1:25.300':
-- In Gold Query: SUBSTRING('1:25.300', 1, 2) -> '1:' (CAST AS INT -> 1) => 60
-- SUBSTRING('1:25.300', 4, 2) -> '5.' (CAST AS INT -> 5) => 65s
-- If a mutant assumes 1-digit minutes (e.g., SUBSTRING(time, 3, 2) for seconds), it gets 25s -> 85s.
-- Let's use '1:59.000': Gold Query gets 1*60 + 9 + 0 = 69s (< 120).
-- What about '2:30.000'? Gold Query: 2*60 + 0 + 0 = 120.0 (Not < 120).
-- What about '1:99.000' or '01:59.999'?
-- Let's test '02:00.005': Gold Query SUBSTRING(time, 7, 2) is '00' -> 120 + 0 + 0/1000 = 120.0.
-- If mutant uses SUBSTRING(time, 7, 3), it gets '005' -> 120.005.
-- To distinguish SUBSTRING(7, 2) vs SUBSTRING(7, 3) around < 120:
-- Suppose seconds is 60 (e.g. '01:59.999' is < 120 for both).
-- What if time is '01:60.000'? No, what if time is '02:00.-50'? No.
-- What if we test 1-digit minute '2:05.000'?
-- Gold Query on '2:05.000': SUBSTRING(1,2)='2:'->2 (120). SUBSTRING(4,2)='5.'->5. Total = 125 (< 120 is False).
-- What about '1:55.000' vs '01:55.000'?
-- Wait: on '01:55.000', mutant calibrated for 1-digit minute (SUBSTRING(1,1)*60 + SUBSTRING(3,2)) gets 0*60 + CAST(':5' AS INT)=0 -> 0s.
-- On '02:10.000', Gold Query gets 2*60 + 10 = 130s (False). A 1-digit minute mutant (SUBSTRING(1,1)*60 + SUBSTRING(3,2)) gets 0*60 + 0 = 0s (< 120 True!).
INSERT INTO lapTimes (raceId, driverId, lap, position, time, milliseconds) VALUES (1, 1, 1, 1, '02:10.000', 90000);
INSERT INTO lapTimes (raceId, driverId, lap, position, time, milliseconds) VALUES (1, 1, 2, 1, '1:30.000', 90000);
