-- DATASET 3
-- Tests handling of NULL in Age and NULLS LAST in Conductor_ID.
-- In SQLite, NULLs are considered smaller than any non-NULL value, so ORDER BY age DESC puts NULL ages at the bottom.
-- If a mutant uses NULLS FIRST on age DESC or sorts ASC, it will fail.
-- Also tests a tie at max age where one Conductor_ID is NULL (allowed by SQLite INTEGER PRIMARY KEY only if not strict, wait: in SQLite, INTEGER PRIMARY KEY cannot be NULL, it auto-generates an ID if NULL is inserted!).
-- So we test negative vs positive Conductor_IDs on tied max Age, and NULL Age.
INSERT INTO conductor (Conductor_ID, Name, Age, Nationality, Year_of_Work) VALUES (1, 'Unknown Age Conductor', NULL, 'France', 50);
INSERT INTO conductor (Conductor_ID, Name, Age, Nationality, Year_of_Work) VALUES (2, 'Senior Conductor B', 80, 'Italy', 30);
