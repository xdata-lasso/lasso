WITH RECURSIVE cnt(x) AS (
  SELECT 1
  UNION ALL
  SELECT x+1 FROM cnt WHERE x < 333
)
INSERT INTO schools (CDSCode, StatusType, County, District, DOC, DOCType, LastUpdate, Phone, Ext)
SELECT CAST(x AS TEXT), 'Active', 'C', 'D', '00', 'T', '2020-01-01', '555-' || CAST(x AS TEXT), CAST(x AS TEXT) FROM cnt;
