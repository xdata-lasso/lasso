```sql
WITH RECURSIVE cnt(x) AS (
  SELECT 1
  UNION ALL
  SELECT x+1 FROM cnt WHERE x < 501
)
INSERT INTO "flights" ("fid", "year", "month_id", "day_of_month", "carrier_id")
SELECT x, 2020, 1, 1, 1 FROM cnt
UNION ALL
SELECT x + 501, 2020, 2, 1, 1 FROM cnt;

INSERT INTO "carriers" ("cid", "name") VALUES (1, 100);
```
