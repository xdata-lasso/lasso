-- many_rows_54: CTE Bulk Order Bounded Quantity Sum with Unique Catalog Fan-Out (complex)
-- Requires SUM(quantity) >= 100000 and MAX(quantity) <= 2 per order (50000 order_items); UNIQUE(order_id, product_id) forces 50000 distinct products with UNIQUE product_name.
WITH bulk_orders AS (SELECT o.order_id, o.customer_id, SUM(oi.quantity) AS total_qty, MAX(oi.quantity) AS max_qty, COUNT(DISTINCT p.product_name) AS distinct_prod_names FROM orders o JOIN order_items oi ON oi.order_id = o.order_id JOIN products p ON p.product_id = oi.product_id WHERE o.order_status = 'PAID' GROUP BY o.order_id, o.customer_id HAVING SUM(oi.quantity) >= 100000 AND MAX(oi.quantity) <= 2) SELECT order_id, customer_id, total_qty, distinct_prod_names FROM bulk_orders;
