q1|Customer Order Summary with Left Join and Status Filtering|SELECT c.customer_id, c.customer_name, COUNT(o.order_id) AS total_orders, COALESCE(SUM(o.total_amount), 0.0) AS total_spend FROM customers c LEFT JOIN orders o ON c.customer_id = o.customer_id AND o.order_status = 'COMPLETED' WHERE c.is_active = 1 GROUP BY c.customer_id, c.customer_name;
q2|Active Engineering Employees without Overdue Tasks|SELECT e.employee_id, e.name, d.dept_name FROM employees e JOIN departments d ON e.dept_id = d.dept_id WHERE e.is_active = 1 AND d.dept_name = 'Engineering' AND NOT EXISTS (SELECT 1 FROM tasks t WHERE t.assigned_to = e.employee_id AND t.status = 'OVERDUE') AND EXISTS (SELECT 1 FROM assignments a JOIN projects p ON a.project_id = p.project_id WHERE a.employee_id = e.employee_id AND p.status = 'ACTIVE');
q3|Top Product per Category with Deterministic Tie Breaking|SELECT product_id, product_name, category_id, list_price FROM (SELECT p.product_id, p.product_name, p.category_id, p.list_price, ROW_NUMBER() OVER (PARTITION BY p.category_id ORDER BY p.list_price DESC, p.product_id ASC) AS rank_in_cat FROM products p WHERE p.is_discontinued = 0) ranked WHERE ranked.rank_in_cat = 1;
